### 3.2 `scripts/generate_known_lenses.rs` (new binary)

```rust
//! Cargo binary: `cargo run --bin generate_known_lenses`  
//! Generates `resources/known_lenses.json` by merging the Lensfun DB
//! with Gyroflow’s own lens‑profile collection.

use anyhow::{Context, Result};
use rayon::prelude::*;
use serde::{Deserialize, Serialize};
use std::{
    collections::{BTreeMap, HashMap},
    env,
    fs,
    path::{Path, PathBuf},
};

/// ---------------------------------------------------------------------------
///  Public data structures – **must stay in sync** with `src/lens/known.rs`
/// ---------------------------------------------------------------------------
#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub struct LensInfo {
    /// Human readable name, e.g. “EF 24‑70mm f/2.8L II USM”
    pub name: String,
    /// Identifier that the stabilization engine expects (same as profile file name)
    pub profile_id: String,
    /// Optional description (focal, aperture, etc.)
    #[serde(skip_serializing_if = "Option::is_none")]
    pub description: Option<String>,
}
pub type ModelMap = BTreeMap<String, Vec<LensInfo>>; // model → lenses
pub type BrandMap = BTreeMap<String, ModelMap>;     // brand → models

#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub struct KnownLenses {
    pub brands: BrandMap,
}

/// ---------------------------------------------------------------------------
///  Lensfun DB structures (mirrors the JSON layout of the Lensfun repo)
/// ---------------------------------------------------------------------------
#[derive(Debug, Clone, Serialize, Deserialize)]
struct LensfunLens {
    maker: String,
    model: String,
    name: String,
    #[serde(default)]
    mount: String,
    #[serde(default, rename = "min_focal")]
    min_focal: f64,
    #[serde(default, rename = "max_focal")]
    max_focal: f64,
    #[serde(default, rename = "min_aperture")]
    min_aperture: f64,
    #[serde(default, rename = "max_aperture")]
    max_aperture: f64,
    #[serde(default, rename = "crop_factor")]
    crop_factor: f64,
    #[serde(rename = "id")]
    lens_id: String,
}

/// ---------------------------------------------------------------------------
///  Minimal Gyroflow lens‑profile meta‑data (extracted from the JSON file)
/// ---------------------------------------------------------------------------
#[derive(Debug, Clone, Serialize, Deserialize)]
struct GyroProfileMeta {
    brand: String,
    model: String,
    lens_name: String,
    #[serde(default)]
    description: Option<String>,
}

/// ---------------------------------------------------------------------------
///  Helper: read every `*.json` file in a directory (non‑recursive) and
///  deserialize it into a map keyed by the *file stem* (profile name).
/// ---------------------------------------------------------------------------
fn read_json_dir<T>(dir: &Path) -> Result<HashMap<String, T>>
where
    T: for<'de> Deserialize<'de>,
{
    let mut map = HashMap::new();
    for entry in fs::read_dir(dir)
        .with_context(|| format!("reading directory {}", dir.display()))?
    {
        let entry = entry?;
        let path = entry.path();
        if path.extension().and_then(|s| s.to_str()) != Some("json") {
            continue;
        }
        let content = fs::read_to_string(&path)
            .with_context(|| format!("reading file {}", path.display()))?;
        let meta: T = serde_json::from_str(&content)
            .with_context(|| format!("parsing JSON {}", path.display()))?;
        let stem = path
            .file_stem()
            .and_then(|s| s.to_str())
            .ok_or_else(|| anyhow::anyhow!("invalid filename {}", path.display()))?
            .to_string();
        map.insert(stem, meta);
    }
    Ok(map)
}

/// ---------------------------------------------------------------------------
///  Load the Lensfun DB (the submodule lives at `external/lensfun-db`).
///  The DB consists of a *cameras* folder and a *lenses* folder – we only need
///  the lenses for the catalog.
/// ---------------------------------------------------------------------------
fn load_lensfun_db(root: &Path) -> Result<Vec<LensfunLens>> {
    let lenses_dir = root.join("lensfun-db").join("data").join("lenses");
    let mut lenses = Vec::new();

    for entry in fs::read_dir(&lenses_dir)
        .with_context(|| format!("reading lensfun lens dir {}", lenses_dir.display()))?
    {
        let entry = entry?;
        let path = entry.path();
        if path.extension().and_then(|s| s.to_str()) != Some("json") {
            continue;
        }
        let content = fs::read_to_string(&path)?;
        // each file contains an array of lenses
        let mut file_lenses: Vec<LensfunLens> = serde_json::from_str(&content)?;
        lenses.append(&mut file_lenses);
    }
    Ok(lenses)
}

/// ---------------------------------------------------------------------------
///  Merge algorithm – Lensfun entries first, then Gyroflow profiles (which
///  may override/extend).  De‑duplicate by `profile_id` while preserving
///  alphabetical order (BTreeMap guarantees it).
/// ---------------------------------------------------------------------------
fn merge(
    lensfun_lenses: &[LensfunLens],
    gyro_profiles: &HashMap<String, GyroProfileMeta>,
) -> KnownLenses {
    let mut brands: BrandMap = BTreeMap::new();

    // 1️⃣ Lensfun entries
    for lf in lensfun_lenses {
        let info = LensInfo {
            name: lf.name.clone(),
            profile_id: lf.lens_id.clone(),
            description: None,
        };
        brands
            .entry(lf.maker.clone())
            .or_default()
            .entry(lf.model.clone())
            .or_default()
            .push(info);
    }

    // 2️⃣ Gyroflow‑specific profiles (may have richer description)
    for (profile_id, meta) in gyro_profiles {
        let info = LensInfo {
            name: meta.lens_name.clone(),
            profile_id: profile_id.clone(),
            description: meta.description.clone(),
        };
        brands
            .entry(meta.brand.clone())
            .or_default()
            .entry(meta.model.clone())
            .or_default()
            .push(info);
    }

    // 3️⃣ De‑duplicate per model (by profile_id) – BTreeMap preserves order
    for model_map in brands.values_mut() {
        for lens_vec in model_map.values_mut() {
            let mut uniq: BTreeMap<&String, LensInfo> = BTreeMap::new();
            for lens in lens_vec.drain(..) {
                uniq.entry(&lens.profile_id).or_insert(lens);
            }
            *lens_vec = uniq.into_values().collect();
        }
    }

    KnownLenses { brands }
}

/// ---------------------------------------------------------------------------
///  Entry point – discovers paths via environment variables (set by CI) or
///  sensible defaults for local development.
/// ---------------------------------------------------------------------------
fn main() -> Result<()> {
    // -------------------------------------------------
    //  1️⃣ Resolve paths
    // -------------------------------------------------
    let repo_root = PathBuf::from(
        env::var("GITHUB_WORKSPACE").unwrap_or_else(|_| env::current_dir()?.to_string_lossy().into()),
    );

    // Lensfun submodule – if missing we bail with a clear message.
    let lensfun_root = repo_root.join("external");
    if !lensfun_root.join("lensfun-db").exists() {
        anyhow::bail!(
            "Lensfun submodule not found at '{}'. Did you forget to run `git submodule update --init`?",
            lensfun_root.display()
        );
    }

    let profiles_dir = repo_root.join("resources").join("lens_profiles");
    let output_file = repo_root.join("resources").join("known_lenses.json");

    // -------------------------------------------------
    //  2️⃣ Load data
    // -------------------------------------------------
    let lensfun_lenses = load_lensfun_db(&repo_root)?;
    let gyro_profiles: HashMap<String, GyroProfileMeta> =
        read_json_dir(&profiles_dir).context("loading Gyroflow lens profiles")?;

    // -------------------------------------------------
    //  3️⃣ Merge → KnownLenses
    // -------------------------------------------------
    let known = merge(&lensfun_lenses, &gyro_profiles);

    // -------------------------------------------------
    //  4️⃣ Serialize with pretty‑print (stable order)
    // -------------------------------------------------
    let json = serde_json::to_string_pretty(&known)?;
    fs::write(&output_file, json)
        .with_context(|| format!("writing output to {}", output_file.display()))?;

    println!("✅  generated {}", output_file.display());
    Ok(())
}
```

*Key points*  

* Uses **Rayon** only for potential parallel loading (currently sequential, but ready for scaling).  
* All paths are resolved relative to `GITHUB_WORKSPACE` (CI) or the current directory (local).  
* Errors are bubbled with `anyhow::Context` for clear diagnostics.  
* Output is deterministic – the JSON is always alphabetically sorted because `BTreeMap` is used throughout.
